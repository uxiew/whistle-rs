# whix example rules — see README.md
# serve a site from a local dev server; the request's remaining path follows
www.example.dev       http://localhost:5173
# map a domain to another address, keeping its Host header (hosts shorthand)
test.local            127.0.0.1:9099
# redirect (short-circuits before contacting upstream)
example.dev/go        redirect://http://test.local/landing
# inject a response header on intercepted traffic (http + https)
example.com           resHeaders://x-mitm=intercepted
# leave a certificate-pinning host alone: no MITM, no capture, bytes relayed as-is
pinned.example.com    sniCallback://no-mitm
